// 1. НАСТРОЙКИ var CONFIG = { streamUrl: "https://lepotafm.ru/listen/lepotafm/radio.mp3", apiUrl: "https://lepotafm.ru/api/nowplaying/lepotafm", statsUrl: "https://script.google.com/macros/s/AKfycbyQsnyXLmGXTNDtL9CLAV6KC80dKm9aIdICEtpY5nqMmldh1gydaPjSc6bozIX8meNWAA/exec", defaultImage: "logo.jpg", refreshTime: 8000 }; // 2. ПЕРЕМЕННЫЕ СОСТОЯНИЯ var isApp = (typeof window.Android !== "undefined"); var audio = new Audio(); var isPlaying = false; window.isPaidUser = false; // 3. ПЕРЕМЕННЫЕ МАЯЧКА var stats = { radioOnly: 0, gameOnly: 0, hybrid: 0, currentGame: "lobby", // Название игры (например: snake, tetris) isGameActive: false // Видит ли юзер игру прямо сейчас }; // 4. СЧЕТЧИК СЕКУНД (тикает раз в секунду) setInterval(function() { // Если игры нет (лобби) ИЛИ она скрыта (юзер в другой вкладке) if (stats.currentGame === "lobby" || stats.isGameActive === false) { if (isPlaying) { stats.radioOnly++; } } // Если юзер ПРЯМО СЕЙЧАС в игре else { if (isPlaying) { stats.hybrid++; // Слушает музыку и играет } else { stats.gameOnly++; // Просто играет в тишине } } }, 1000); // 5. ОТПРАВКА В ТАБЛИЦУ (каждые 15 секунд для теста) setInterval(sendBeacon, 15000); function sendBeacon() { if (stats.radioOnly === 0 && stats.gameOnly === 0 && stats.hybrid === 0) return; var query = "radio_only=" + stats.radioOnly + "&game_only=" + stats.gameOnly + "&hybrid=" + stats.hybrid + "&last_game=" + encodeURIComponent(stats.currentGame) + "&user_type=" + (window.isPaidUser ? "paid" : "free"); fetch(CONFIG.statsUrl, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: query }); // Очистка счетчиков stats.radioOnly = 0; stats.gameOnly = 0; stats.hybrid = 0; } // 6. УПРАВЛЕНИЕ ВКЛАДКАМИ (Табами) window.openTab = function(tabName, btn) { // Скрываем все вкладки var panes = document.querySelectorAll('.tab-pane'); for (var i = 0; i < panes.length; i++) { panes[i].classList.remove('active'); } // Убираем подсветку кнопок var btns = document.querySelectorAll('.tab-btn'); for (var j = 0; j < btns.length; j++) { btns[j].classList.remove('active'); } // Показываем нужную var target = document.getElementById(tabName); if (target) target.classList.add('active'); if (btn) btn.classList.add('active'); // ЛОГИКА МАЯЧКА: // Если перешли в 'home', значит видим игру if (tabName === 'home') { var frame = document.getElementById('game-frame'); // Если во фрейме реально что-то есть (не пусто), считаем игру активной if (frame && frame.src && !frame.src.includes('about:blank')) { stats.isGameActive = true; } else { stats.isGameActive = false; stats.currentGame = "lobby"; } } else { // Если ушли в настройки, историю и т.д. - игра на паузе stats.isGameActive = false; } }; // 7. ЗАГРУЗКА ИГРЫ window.loadGame = function(gamePath) { // Вычисляем имя игры из пути (из games/snake.html получим snake) try { var parts = gamePath.split('/'); var fileName = parts[parts.length - 1]; stats.currentGame = fileName.replace('.html', ''); } catch(e) { stats.currentGame = "unknown"; } stats.isGameActive = true; var frame = document.getElementById('game-frame'); if (frame) { // Добавляем v=время, чтобы не было кэша и черного экрана var sep = gamePath.indexOf('?') !== -1 ? '&' : '?'; frame.src = gamePath + sep + "v=" + Date.now(); // Переходим на главную вкладку, где живет фрейм window.openTab('home', document.querySelector('.tab-btn[onclick*="home"]')); } }; // 8. ПЛЕЕР И ГРОМКОСТЬ function initPlayer() { var playBtn = document.getElementById('play-btn'); if (playBtn) playBtn.addEventListener('click', togglePlayState); } function togglePlayState() { var slider = document.getElementById('vol-slider'); if (isApp && window.Android) { if (isPlaying) { window.Android.pauseAudio(); isPlaying = false; } else { if (slider) window.Android.setVolume(parseFloat(slider.value)); window.Android.playAudio(); isPlaying = true; } } else { if (isPlaying) { audio.pause(); audio.src = ""; isPlaying = false; } else { audio.src = CONFIG.streamUrl + "?n=" + Date.now(); audio.play().catch(function(){}); isPlaying = true; } } // Обновляем иконки window.syncAppState(isPlaying, window.isPaidUser); } function initVolume() { var s = document.getElementById('vol-slider'); if (!s) return; var v = localStorage.getItem('savedVolume') || 1.0; s.value = v; if (isApp && window.Android) window.Android.setVolume(parseFloat(v)); else audio.volume = v; s.oninput = function(e) { var vol = e.target.value; localStorage.setItem('savedVolume', vol); if (isApp && window.Android) window.Android.setVolume(parseFloat(vol)); else audio.volume = vol; }; } // 9. СИНХРОНИЗАЦИЯ (для Android) window.syncAppState = function(p, paid) { isPlaying = p; window.isPaidUser = paid; var icon = document.getElementById('play-icon'); var playerDiv = document.querySelector('.inline-player'); if (icon) icon.className = isPlaying ? "fas fa-pause" : "fas fa-play"; if (playerDiv) isPlaying ? playerDiv.classList.add('playing') : playerDiv.classList.remove('playing'); }; // 10. ЗАГРУЗКА СТРАНИЦЫ window.onload = function() { if (!isApp) { audio.src = CONFIG.streamUrl; } else { if(window.Android && window.Android.notifyPageLoaded) window.Android.notifyPageLoaded(); } initPlayer(); initVolume(); updateMetadata(); setInterval(updateMetadata, CONFIG.refreshTime); }; function updateMetadata() { fetch(CONFIG.apiUrl + "?t=" + Date.now()).then(function(r){return r.json()}).then(function(d){ if (d.now_playing && d.now_playing.song) { var s = d.now_playing.song; if(document.getElementById('track-name')) document.getElementById('track-name').innerText = s.title; if(document.getElementById('artist-name')) document.getElementById('artist-name').innerText = s.artist; if(document.getElementById('mini-art')) document.getElementById('mini-art').src = s.art.replace('http:','https:'); } }).catch(function(){}); }
